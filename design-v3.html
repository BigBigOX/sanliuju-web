3. 模块接口规范

3.1 新增模块：engine/room-manager.js

// 房间管理 — V3 新增模块
function createRoomManager(stateManager) {
  const S = stateManager.state

  function createRoom(config) {
    // config: { maxPlayers, theme, roundCount }
    S.room = {
      id: genRoomId(),        // 6位数字
      status: 'waiting',      // waiting | playing | ended
      round: 1,
      maxPlayers: config.maxPlayers || 6,
      theme: config.theme || 'mixed',
      totalRounds: config.roundCount || 3,
      createdAt: Date.now()
    }
    return S.room
  }

  function canStart() {
    const joined = S.players.filter(p => p.joined && !p.isHost)
    return joined.length >= 2 && joined.length <= S.room.maxPlayers
  }

  function getRoomSummary() {
    // 玩家端看到的大厅信息
    return {
      roomId: S.room.id,
      status: S.room.status,
      theme: S.room.theme,
      totalRounds: S.room.totalRounds,
      playerCount: S.players.filter(p => p.joined && !p.isHost).length,
      hostName: S.players.find(p => p.isHost)?.nickname || '主持人'
    }
  }

  return { createRoom, canStart, getRoomSummary }
}

9. 安全与内容审核

9.1 多层审核架构

[用户输入] → [前端预处理] → [微信 msgSecCheck] → [腾讯云 IMS] → [通过/拒绝] ↓ ↓ [审核日志 (Supabase)] 前端预处理： - 敏感词本地过滤（正则+词库） - 长度限制（昵称 ≤8字，回答 ≤50字） - 空值/重复检测 微信 msgSecCheck (仅小程序)： - 调用 wx.cloud.callFunction 或直接 API - 返回: PASS / REVIEW / BLOCK 腾讯云 IMS (Web + 小程序后端兜底)： - 文本内容安全检测 - 返回: Normal / Porn / Abuse / Ad / Illegal 等标签 - 建议阈值: 任何非 Normal 标签 → REVIEW → 人工 审核日志 (Supabase audit_logs): - 记录 content_type, content, check_result, timestamp - 保留 90 天 - REVIEW 结果 24h 内人工处理

9.2 小程序文本安全检测流程

// 玩家提交回答
async function submitAnswer(text) {
  // Step 1: 前端预处理
  const trimmed = text.trim().slice(0, 50)
  if (!trimmed || trimmed.length < 1) return toast('请输入回答')

  // Step 2: 微信安全检测 (小程序)
  if (PLATFORM === 'miniapp') {
    const checkResult = await wx.cloud.callFunction({
      name: 'contentCheck',
      data: { content: trimmed, scene: 1 }
    })
    if (checkResult.result.suggest === 'risky') {
      return toast('内容不合规，请重新输入')
    }
    // 记录审核日志
    logAudit(S.playerId, 'answer', trimmed, checkResult)
  }

  // Step 3: 广播回答
  Channel.broadcast({ type: 'PLAYER_ANSWER', ... })
}
